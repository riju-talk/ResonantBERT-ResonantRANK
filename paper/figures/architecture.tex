% token-state row: [CLS] h h [SEP] b b b [SEP]; #1 = x of first cell, #2 = y
\newcommand{\tokrow}[2]{%
  \foreach \c [count=\i from 0] in {black!30,hdC,hdC,black!30,bdC,bdC,bdC,black!30}
    \fill[\c] (#1+\i*0.15-0.065,#2-0.065) rectangle ++(0.13,0.13);}

% value badge: \badge{cx}{top y}{icon}{name}{level in [0,5]}{colour}
\newcommand{\badge}[6]{%
  \draw[#6!75!black, fill=white, line width=0.45pt, rounded corners=2pt] (#1-0.70,#2-1.0) rectangle (#1+0.70,#2);
  \node[text=#6!80!black, inner sep=0pt] at (#1,#2-0.25) {\fontsize{9}{9}\selectfont #3};
  \node[font=\fontsize{6.3}{7}\selectfont, inner sep=0pt] at (#1,#2-0.54) {#4};
  \foreach \k in {0,...,4}{%
    \pgfmathsetmacro{\xa}{#1-0.465+\k*0.19}%
    \pgfmathsetmacro{\w}{max(0,min(1,#5-\k))*0.16}%
    \fill[#6] (\xa,#2-0.86) rectangle ++(\w,0.13);%
    \draw[#6!70!black, line width=0.3pt] (\xa,#2-0.86) rectangle ++(0.16,0.13);}}

% candidate row in a list: \cand{x}{y}{score in [0,1]}{colour}
\newcommand{\cand}[4]{%
  \draw[black!55, fill=white, line width=0.4pt] (#1,#2-0.11) -- ++(0,0.22) -- ++(0.12,0) -- ++(0.05,-0.05) -- ++(0,-0.17) -- cycle;
  \fill[#4] (#1+0.25,#2-0.06) rectangle ++(#3*0.72,0.12);}
